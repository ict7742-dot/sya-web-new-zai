    45→function captureUtm(): UtmData {
    46→  if (typeof window === 'undefined') return { utmSource: '', utmMedium: '', utmCampaign: '', utmTerm: '', utmContent: '', landingPage: '' };
    47→  const params = new URLSearchParams(window.location.search);
    48→  const data: UtmData = {
    49→    utmSource: params.get('utm_source') || '',
    50→    utmMedium: params.get('utm_medium') || '',
    51→    utmCampaign: params.get('utm_campaign') || '',
    52→    utmTerm: params.get('utm_term') || '',
    53→    utmContent: params.get('utm_content') || '',
    54→    landingPage: window.location.href.split('?')[0],
    55→  };
    56→  if (data.utmSource || data.utmMedium || data.utmCampaign) {
    57→    sessionStorage.setItem('sya_utm', JSON.stringify(data));
    58→  }
    59→  return data;
    60→}
    61→
    62→function getStoredUtm(): UtmData {
    63→  if (typeof window === 'undefined') return { utmSource: '', utmMedium: '', utmCampaign: '', utmTerm: '', utmContent: '', landingPage: '' };
    64→  try {
    65→    const raw = sessionStorage.getItem('sya_utm');
    66→    if (raw) return JSON.parse(raw) as UtmData;
    67→  } catch { /* ignore */ }
    68→  return { utmSource: '', utmMedium: '', utmCampaign: '', utmTerm: '', utmContent: '', landingPage: window.location.href.split('?')[0] };
    69→}

    71→/* Form abandonment — saves partial form data to localStorage */
    72→const ABANDON_KEY = 'sya_form_abandon';
    73→
    74→function saveAbandonment(data: Record<string, string>) {
    75→  try {
    76→    localStorage.setItem(ABANDON_KEY, JSON.stringify({ ...data, savedAt: new Date().toISOString() }));
    77→  } catch { /* ignore */ }
    78→}
    79→
    80→function clearAbandonment() {
    81→  try { localStorage.removeItem(ABANDON_KEY); } catch { /* ignore */ }
    82→}
    83→
    84→function loadAbandonment(): Record<string, string> | null {
    85→  try {
    86→    const raw = localStorage.getItem(ABANDON_KEY);
    87→    if (!raw) return null;
    88→    const data = JSON.parse(raw);
    89→    // Only restore if less than 7 days old
    90→    const age = Date.now() - new Date(data.savedAt).getTime();
    91→    if (age > 7 * 24 * 60 * 60 * 1000) { clearAbandonment(); return null; }
    92→    return data;
    93→  } catch { return null; }
    94→}

   266→function mulberry32(a: number) {
   267→  return function () {
   268→    a |= 0;
   269→    a = (a + 0x6d2b79f5) | 0;
   270→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   271→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   272→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   273→  };
   274→}

   276→function inr(v: number, d = 2) {
   277→  if (v == null || isNaN(v)) return '—';
   278→  return v.toLocaleString('en-IN', {
   279→    minimumFractionDigits: d,
   280→    maximumFractionDigits: d,
   281→  });
   282→}

   284→function genSeries(s: SymbolData): SeriesData {
   285→  const rnd = mulberry32(s.seed);
   286→  const n = 44;
   287→  const arr: Candle[] = [];
   288→  let p = s.start;
   289→  for (let i = 0; i < n; i++) {
   290→    const o = p;
   291→    const c = o * (1 + 0.0007 + (rnd() - 0.5) * s.vol * 2);
   292→    const h = Math.max(o, c) * (1 + rnd() * s.vol * 0.9);
   293→    const l = Math.min(o, c) * (1 - rnd() * s.vol * 0.9);
   294→    arr.push({ o, h, l, c });
   295→    p = c;
   296→  }
   297→  return {
   298→    candles: arr,
   299→    prevClose: s.start * 0.994,
   300→    volM: 90 + rnd() * 80,
   301→    tickCount: 0,
   302→  };
   303→}

   307→function LazySection({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   308→  const [visible, setVisible] = useState(false);
   309→  const ref = useRef<HTMLDivElement>(null);
   310→  useEffect(() => {
   311→    const el = ref.current;
   312→    if (!el) return;
   313→    const observer = new IntersectionObserver(
   314→      ([entry]) => { if (entry.isIntersecting) { setVisible(true); observer.unobserve(el); } },
   315→      { rootMargin: '200px' }
   316→    );
   317→    observer.observe(el);
   318→    return () => observer.disconnect();
   319→  }, []);
   320→  return (
   321→    <div ref={ref} className={className}>
   322→      {visible ? children : <div className="sec"><div className="wrap"><div style={{ height: '300px' }} className="animate-pulse bg-white/[0.01] rounded-xl" /></div></div>}
   323→    </div>
   324→  );
   325→}
